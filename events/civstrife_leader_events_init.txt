country_event = { # Commander
	id = civleader_init.1000
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		any_owned_leader = {
			OR = {
				has_base_skill >= 5
				has_leader_flag = leader_ambitious
				has_leader_flag = leader_great_victory
				has_leader_flag = leader_proven_worth
			}
		}
	}

	immediate = {
		every_owned_leader = {
			limit = {
				OR = {
					has_base_skill >= 5
					has_leader_flag = leader_ambitious
					has_leader_flag = leader_great_victory
					has_leader_flag = leader_proven_worth
				}
			}
			random = {
				chance = value:leader_gains_power
			}
		}
	}
}
