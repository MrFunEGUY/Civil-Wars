leader_event = { ## Declare War or Not Decision
	id = civleader_init.1090
	title = civleader_init.1090.name
	picture = "GFX_evt_enclave_overlord"
	is_triggered_only = yes

	desc = {
		text = civleader_init.1090.dem.desc
		trigger = { owner = { has_democratic_elections = yes } }
	}

	desc = {
		text = civleader_init.1090.not_dem.desc
		trigger = { owner = { has_democratic_elections = no } }
	}
	
	immediate = {
		random = {
			chance = value:election_out_is_allowed
			# chance = 100
			set_leader_flag = leader_can_be_pacified
		}
	}

	option = {
		name = {
			trigger = { owner = { has_democratic_elections = yes } }
			text = cw_leader_new_election
		}
		name = {
			trigger = { owner = { has_democratic_elections = no } }
			text = cw_leader_give_in
		}
		trigger = { has_leader_flag = leader_can_be_pacified }
		if = {
			limit = { owner = { has_democratic_elections = no } }
			owner = { set_leader = root }
		}
		else = {
			custom_tooltip = cw_leader_new_election_tooltip
			## Only way to call elections rn is to kill ruler, so we will create dummy leader, set them ruler, kill them
			hidden_effect = {
				owner = {
					create_leader = {
						class = official
						species = owner_main_species
						name = random
						effect = {
							save_event_target_as = dummy_election_starter_leader
						}
					}
					set_leader = event_target:dummy_election_starter_leader
					event_target:dummy_election_starter_leader = { kill_leader = { show_notification = no } }
				}
			}
		}
		hidden_effect = {
			remove_leader_flag = leader_can_be_pacified
		}
	}

	option = {
		name = cw_leader_ignore_them
		hidden_effect = {
			random = {
				chance = value:leader_dispute_civil_war_chance
				# chance = 100
				create_civil_war = yes
			}
		}
	}

	after = {
		hidden_effect = {
			set_cooldown = 0
			remove_leader_flag = cw_cannot_be_fired
			if = {
				limit = { NOT = { has_leader_flag = personal_civil_war } }
				leader_event = { id = civleader_init.1091 days = 6 random = 14 }
			}

			if = {
				limit = { exists = event_target:new_elections_leader }
				set_owner = event_target:exile_leader_home
			}
		}
	}
}
