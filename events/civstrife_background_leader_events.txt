namespace = civ_misc_leader

leader_event = { # Kill Dummy Leaders
	id = civ_misc_leader.1000
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		exists = this
	}

	immediate = {
		if = {
			limit = { NOT = { has_leader_flag = dummy_leader } }
			set_dummy_leader_flags = yes
		}
		kill_leader = { show_notification = no }
	}
}

country_event = { # Set Corrupt Official Flag
	id = civ_misc_leader.2000
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		is_country_type_with_subjects = yes
		any_owned_leader = {
			has_trait = leader_trait_corrupt
			NOT = { has_leader_flag = corrupt_official }
		}
	}

	immediate = {
		every_owned_leader = {
			limit = {
				has_trait = leader_trait_corrupt
				NOT = { has_leader_flag = corrupt_official }
			}
			set_leader_flag = corrupt_official
		}
	}
}

# on_ruler_set
# Executed as new ruler has been set
# This = Country
country_event = { # Set Corrupt Ruler Trait
	id = civ_misc_leader.2005
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		ruler = {
			has_leader_flag = corrupt_official
			NOT = { has_trait = leader_trait_corrupt }
		}
	}

	immediate = {
		ruler = {
			add_trait = {
				trait = leader_trait_corrupt
				show_message = no
			}
		}
	}
}

# on_ruler_set
# Executed as new ruler has been set
# This = Country
country_event = { # Set Sep Won Election Flag
	id = civ_misc_leader.2100
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		ruler = {
			pop_faction? = { is_separatist_faction = yes }
		}
	}

	immediate = {
		ruler = { set_timed_leader_flag = { flag = sep_won_election years = 7 } }
	}
}

country_event = {
	id = civ_misc_leader.3000
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		ruler = {
			OR = {
				has_leader_flag = traitor_@root
				has_leader_flag = heir_traitor_@root
			}
			OR = {
				has_trait = leader_trait_traitor
				has_trait = leader_trait_traitor_heir
			}
		}
	}

	immediate = {

		ruler = {
			export_trigger_value_to_variable = {
				trigger = years_passed
				variable = year_restored
			}

			change_variable = { which = year_restored value = 2200 }

			set_variable = { which = years_exiled value = year_restored }

			if = {
				limit = { has_trait = leader_trait_traitor }
				subtract_variable = { which = years_exiled value = year_fled }

				remove_trait = leader_trait_traitor
				add_trait = {
					trait = leader_trait_vindicated_revolutionary
					show_message = no
				}
			}

			if = {
				limit = { has_trait = leader_trait_traitor_heir }
				subtract_variable = { which = years_exiled value = year_disinherited }

				remove_trait = leader_trait_traitor_heir
				add_trait = {
					trait = leader_trait_vindicated_heir
					show_message = no
				}
			}
		}
	}
}
