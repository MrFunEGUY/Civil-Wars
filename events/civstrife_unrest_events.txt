namespace = civstrife_unrest

@unrest_event_cooldown = 360

situation_event = {
	id = civstrife_unrest.100
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		situation_progress > 0
		NOT = { has_situation_flag = unrest_ethic_event }
	}

	immediate = {
		set_situation_flag = unrest_ethic_event
		random = { ## This doesnt always need to fire
			chance = 55
			target = {
				random_owned_pop_group = { planet = { save_event_target_as = random_rebel_planet } }
				event_target:random_rebel_planet = {
					set_timed_planet_flag = {
						flag = recent_unrest_event
						days = @unrest_event_cooldown
					}
					planet_event = { id = civstrife_unrest.101 }
				}
			}
		}
	}
}
