### Civil Wars

# Install Traitor
war_name_format = {
	random_weight = {
		factor = 0
		modifier = {
			add = 999
			using_war_goal = {
				type = wg_install_traitor
				owner = attacker
			}
		}
	}
	format = "The {[This.MainAttacker.GetAdj] {<installation_war_names>}}"
}

# Retrieve Traitor
war_name_format = {
	random_weight = {
		factor = 0
		modifier = {
			add = 999
			OR = {
				using_war_goal = {
					type = wg_retrieve_traitor
					owner = attacker
				}
				using_war_goal = {
					type = wg_retrieve_traitor_heir
					owner = attacker
				}
			}
		}
	}
	format = "{<retrieval_war_names> {[This.MainAttacker.GetAdj]}}"
}
