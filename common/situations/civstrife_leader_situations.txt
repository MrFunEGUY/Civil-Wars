
# Separatist Planet
situation_civstrife_election_dispute = {
	title = {
		trigger = { owner = { is_democratic_authority = yes } }
		text = election_dispute_dem_name
	}
	title = {
		trigger = { owner = { is_democratic_authority = no } }
		text = election_dispute_not_dem_name
	}
	desc = {
		trigger = { owner = { is_democratic_authority = yes } }
		text = election_dispute_dem_desc
	}
	desc = {
		trigger = { owner = { is_democratic_authority = no } }
		text = election_dispute_not_dem_desc
	}
	picture = GFX_evt_two_sided_deal
	category = positive

	fail_icon = GFX_situation_outcome_negative
	fail_icon_frame = GFX_situation_outcome_frame_red
	complete_icon = GFX_situation_outcome_positive
	complete_icon_frame = GFX_situation_outcome_frame_green

	on_start = {
		owner = {

			export_trigger_value_to_variable = {
				trigger = resource_revenue_compare
				parameters = { resource = energy }
				variable = monthly_energy
			}

			export_trigger_value_to_variable = {
				trigger = resource_revenue_compare
				parameters = { resource = unity }
				variable = monthly_unity
			}
		}
	}

	on_progress_complete = {
		if = {
			limit = { exists = target }
			target = {
				remove_leader_flag = cw_cannot_be_fired
				leader_event = { id = civleader.1090 }
			}
		}
		hidden_effect = {
			owner = {
				clear_variable = monthly_energy
				clear_variable = monthly_unity
			}
			destroy_situation = this
		}
	}

	on_fail = { }

	on_abort = {
		owner = {
			clear_variable = monthly_energy
			clear_variable = monthly_unity
		}
		if = {
			limit = { exists = target }
			target = { remove_leader_flag = cw_cannot_be_fired }
		}
		owner = { country_event = { id = civleader.1003 } }
	}

	abort_trigger = {
		OR = {
			owner = { num_owned_planets < 2 }
			NOT = { exists = target }
			NOT = { target = { is_owned_by = root.owner } }
		}
	}

	start_value = 0

	stages = {
		election_stage_1 = {
			icon = GFX_situation_stage_1
			icon_background = GFX_situation_stage_frame_red
			modifier = { planet_stability_add = -2.5 }
			end = 100
		}
	}

	approach = {
		name = approach_election_do_nothing
		icon = GFX_situation_approach_scales
		icon_background = GFX_situation_approach_bg_green
		default = yes
		allow = { NOT = { current_situation_approach = approach_election_obstruct } }
		on_select = { custom_tooltip = election_nothing_tooltip }
		resources = {
			category = situations
			upkeep = {
				energy = 0.05
				multiplier = owner.monthly_energy
			}
			upkeep = {
				unity = 0.05
				multiplier = owner.monthly_unity
			}
		}
		ai_weight = {
			base = 5
		}
	}

	approach = {
		name = approach_election_obstruct
		icon = GFX_situation_approach_fist
		icon_background = GFX_situation_approach_bg_yellow
		on_select = { custom_tooltip = election_obstruction_tooltip }
		resources = {
			category = situations
			upkeep = {
				energy = 0.15
				multiplier = owner.monthly_energy
			}
			upkeep = {
				unity = 0.15
				multiplier = owner.monthly_unity
			}
		}
		modifier = { planet_stability_add = -5 }
		ai_weight = {
			base = 2
		}
	}

	on_monthly = {
		events = {
			# civelect.1050			# Random Events
	    }
	    random_events = {
			85 = 0
			10 = civleader.1010		## Protests
			3 = civleader.1011		## Terror
			2 = civleader.1020		## Watergate
		}
	}

	monthly_progress = {
		base = 15

		modifier = {
			add = -5
			desc = obstruction_hindering_investigation
			current_situation_approach = approach_election_obstruct
		}
	}
}
