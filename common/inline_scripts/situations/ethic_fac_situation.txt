on_progress_complete = {
	custom_tooltip = situation_civstrife_separatist_end
	hidden_effect = {
		if = {
			limit = { exists = target }
			target = {
				remove_pop_faction_flag = faction_sep_situation
			}
		}
	}
}

on_fail = {
	custom_tooltip = situation_civstrife_separatist_failed
	hidden_effect = {
		if = {
			limit = { exists = target }
			target = {
				remove_pop_faction_flag = faction_sep_situation
			}
		}
	}
}

on_abort = {
	if = {
		limit = { exists = target }
		target = {
			remove_pop_faction_flag = faction_sep_situation
		}
	}
}

abort_trigger = {
	OR = {
		situation_progress < 0
		target = { has_pop_faction_flag = faction_rebelled }
		NOT = { exists = target }
		owner = {
			OR = {
				num_owned_planets < 2
				has_country_flag = civ_war_recent_$FACTION$
				these_are_rebels_in_rebellion = yes
			}
		}
	}
}
