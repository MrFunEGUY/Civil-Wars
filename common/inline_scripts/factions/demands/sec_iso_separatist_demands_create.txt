# Independence
demand = {
	title = "SEPARATISM_DEMAND_IND_s_x"
	unfulfilled_title = "SEPARATISM_DEMAND_IND_s_x"
	desc = "SEPARATISM_DEMAND_IND_s_iso_$ISO_SEC_FAC_NUM$_desc"

	unfulfilled_effect = -30
	fulfilled_effect = 0.001

	potential = {
		exists = event_target:sec_iso_separatist_$ISO_SEC_FAC_NUM$
	}

	trigger = {
		owner = {
			NOR = {
				any_owned_sector = {
					is_same_value = event_target:sec_iso_separatist_$ISO_SEC_FAC_NUM$
				}
			}
		}
	}
}

# Federal Government Policy - Ethics and Civics
demand = {
	title = "SEPARATISM_DEMAND_FEDERAL_s_x"
	unfulfilled_title = "SEPARATISM_DEMAND_FEDERAL_s_x"
	desc = "SEPARATISM_DEMAND_FEDERAL_s_$ISO_SEC_FAC_NUM$_desc"

	unfulfilled_effect = -10
	fulfilled_effect = 5

	potential = {
		has_global_flag = cultural_overhaul_active
		exists = event_target:sec_iso_separatist_$ISO_SEC_FAC_NUM$
	}

	trigger = {
		owner = { has_policy_flag = federal_state }
	}
}

on_create = {
	event_target:sec_iso_separatist_$ISO_SEC_FAC_NUM$ = { save_event_target_as = sep_target_loc }
	sector_faction_first_effect = yes
	set_name = {
		key = iso_sector_separatist_real_name
		variable_string = "[last_created_leader.GetName]"
		variable_string = "[sep_target_loc.GetName]"
	}
	sector_faction_second_effect = yes
}

on_destroy = {
	# Allows planets to resume getting separatists, which in turn will allow this to re-form if necessary
	# faction_kill_leader_effect = yes
	if = {
		limit = { exists = event_target:sec_iso_separatist_$ISO_SEC_FAC_NUM$ }
		event_target:sec_iso_separatist_$ISO_SEC_FAC_NUM$ = { remove_sector_flag = sector_sep_active }
		clear_global_event_target = sec_iso_separatist_$ISO_SEC_FAC_NUM$
	}
}
