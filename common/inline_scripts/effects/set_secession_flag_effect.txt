limit = { event_target:global_event_country = { check_variable = { which = secession_country_count value = $NUM$ } } }
event_target:civwar_originator = {
	set_country_flag = secession_country_$NUM$_flag
	save_global_event_target_as = secession_country_$NUM$
	set_country_flag = country_needed_for_loc
}
event_target:civwar_rebels = {
	set_country_flag = secession_country_$NUM$_seceder
	set_country_flag = secession_from_country_@event_target:civwar_originator
	if = {
		limit = { exists = ruler }
		ruler = { set_leader_flag = secession_country_$NUM$_traitor }
	}
}
