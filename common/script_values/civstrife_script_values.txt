	base = @base_leader_event_chance

	inline_script = { script = script_values/leader_standard }

	# Ethic
	modifier = { is_pacifist = yes factor = 0.5 }
	
	modifier = { owner = { is_democratic_authority = yes }	factor = 0.33 }
	modifier = { owner = { is_imperial_authority = yes }	factor = 1 }
	modifier = { owner = { is_dictatorial_authority = yes }	factor = 1.25 }
	modifier = { owner = { is_oligarchic_authority = yes }	factor = 1.5 }

	## Traits
	modifier = { has_trait = leader_trait_corrupt add = 5 }

	## War Values
	modifier = {
		add = 1
		exists = solar_system
		solar_system = { is_capital_system = no }
		owner.capital_scope = { is_occupied_flag = yes }
	}

	modifier = {
		factor = 0.25
		exists = planet
		planet = { is_occupied_flag = yes }
	}

	modifier = {
		factor = @system_occupation_crew_mult
		exists = planet.starbase
		NOT = { planet.starbase.controller = { is_same_value = prev.planet.starbase.owner } }
	}

	modifier = {
		factor = planet.value:ecaf_crew_devastation_mult
		planet = { planet_devastation > 0 }
	}

	modifier = {
		factor = owner.value:ecaf_crew_att_exhaustion_mult
		owner = {
			is_at_war = yes
			any_war = { any_attacker = {  is_same_value = prevprev  } }
		}
	}

	modifier = {
		factor = owner.value:ecaf_crew_def_exhaustion_mult
		owner = {
			is_at_war = yes
			any_war = { any_defender = {  is_same_value = prevprev  } }
		}
	}

	modifier = {
		factor = owner.value:ecaf_crew_bonus_mult
		owner = {
			pop_amount < 5000
			OR = {
				is_rogue_colony = yes
				has_civstrife_origin = yes
			}
		}
	}

	modifier = {
		add = 500000
		has_global_flag = ecaf_crew_cheat_enabled
	}
}
