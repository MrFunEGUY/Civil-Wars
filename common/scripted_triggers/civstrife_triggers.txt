frontier_planet_slave_tick_check = {
	has_any_planet_rebellion_stopper = no
	planet_stability < 25
	has_planet_flag = frontier_planet_@root
	NOT = { planet_devastation >= 30 }
	NOT = { any_targeting_situation = { is_situation_type = situation_civstrife_sep_plan } }
	if = {
		limit = { exists = sector }
		sector = {
			NOT = { has_sector_flag = sector_sep_active }
			NOT = { has_sector_flag = sec_slave_sep_active }
		}
	}
	count_owned_pop_amount = {
		limit = {
			can_think = yes
			pop_group_has_happiness = yes
		}
		count >= 200
	}
	pop_amount_percentage = {
		percentage >= 0.3
		limit = {
			is_enslaved_or_purged = yes
			happiness < 0.8
		}
		exclude = { can_think = no }
	}
}

		is_homicidal = no

		}
		AND = {
			is_organic_zealots = yes

		}
	}
	NOR = {

	}

	is_country_type_with_subjects = yes

	}
	NOT = { is_same_value = event_target:civwar_rebels }
	NOT = { has_truce = event_target:civwar_originator }
	NOT = { has_overlord = event_target:civwar_originator }
	NOT = { is_same_empire = event_target:civwar_originator }
	NOT = { has_defensive_pact = event_target:civwar_originator }
	NOT = { is_in_federation_with = event_target:civwar_originator }
	NOT = { has_association_status = event_target:civwar_originator }
	NOT = { has_non_aggression_pact = event_target:civwar_originator }
	if = {
		limit = {
			exists = event_target:civwar_originator.overlord
			event_target:civwar_rebels = { is_at_war_with = event_target:civwar_originator.overlord }

		NOT = { has_truce = event_target:civwar_originator.overlord }
