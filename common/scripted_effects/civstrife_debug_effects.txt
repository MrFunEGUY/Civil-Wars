get_civwar_countries = {
	if = {
		limit = { any_playable_country = { is_rival_government = yes } }
		log = "Rival Government:"
		every_playable_country = {
			limit = { is_rival_government = yes }
			log = "\\[this.GetName]"
		}
		log = "-------------------------"
	}
	if = {
		limit = { any_playable_country = { has_origin = origin_civwar } }
		log = "Separatist State:"
		every_playable_country = {
			limit = { has_origin = origin_civwar }
			log = "\\[this.GetName]"
		}
		log = "-------------------------"
	}
	if = {
		limit = { any_playable_country = { has_origin = origin_nationalist_rebellion } }
		log = "Nationalist Uprising:"
		every_playable_country = {
			limit = { has_origin = origin_nationalist_rebellion }
			log = "\\[this.GetName]"
		}
		log = "-------------------------"
	}
	if = {
		limit = { any_playable_country = { has_origin = origin_plan_rebellion } }
		log = "Planetary Rebellion:"
		every_playable_country = {
			limit = { has_origin = origin_plan_rebellion }
			log = "\\[this.GetName]"
		}
		log = "-------------------------"
	}
	if = {
		limit = { any_playable_country = { has_origin = origin_civstrife_slave_rebellion } }
		log = "Slave Uprising:"
		every_playable_country = {
			limit = { has_origin = origin_civstrife_slave_rebellion }
			log = "\\[this.GetName]"
		}
		log = "-------------------------"
	}
	if = {
		limit = { any_playable_country = { has_origin = origin_heir_rebellion } }
		log = "Disinherited Heir:"
		every_playable_country = {
			limit = { has_origin = origin_heir_rebellion }
			log = "\\[this.GetName]"
		}
		log = "-------------------------"
	}
	if = {
		limit = { any_playable_country = { has_origin = origin_sec_rebellion } }
		log = "Sectoral Rebellion:"
		every_playable_country = {
			limit = { has_origin = origin_sec_rebellion }
			log = "\\[this.GetName]"
		}
		log = "-------------------------"
	}
	if = {
		limit = { any_playable_country = { has_origin = origin_isolated_rebellion } }
		log = "Isolated Territory:"
		every_playable_country = {
			limit = { has_origin = origin_isolated_rebellion }
			log = "\\[this.GetName]"
		}
		log = "-------------------------"
	}
	if = {
		limit = { any_playable_country = { has_origin = origin_coup_successor } }
		log = "Disputed Successor:"
		every_playable_country = {
			limit = { has_origin = origin_coup_successor }
			log = "\\[this.GetName]"
		}
		log = "-------------------------"
	}
	if = {
		limit = { any_playable_country = { has_origin = origin_coup_independent } }
		log = "Disputed Fealty:"
		every_playable_country = {
			limit = { has_origin = origin_coup_independent }
			log = "\\[this.GetName]"
		}
		log = "-------------------------"
	}
	if = {
		limit = { any_playable_country = { has_origin = origin_civsecession } }
		log = "Political Secession:"
		every_playable_country = {
			limit = { has_origin = origin_civsecession }
			log = "\\[this.GetName]"
		}
		log = "-------------------------"
	}
	if = {
		limit = {
			any_playable_country = {
				OR = {
					has_origin = origin_exile_government
					has_origin = origin_disinherited_heir
					has_origin = origin_restored_government
					has_origin = origin_restored_successor_government
				}
			}
		}
		log = "========================"
	}
	if = {
		limit = { any_playable_country = { has_origin = origin_exile_government } }
		log = "Exiled Government:"
		every_playable_country = {
			limit = { has_origin = origin_exile_government }
			log = "\\[this.GetName]"
		}
		log = "-------------------------"
	}

		limit = { any_playable_country = { has_origin = origin_disinherited_heir } }

			limit = { has_origin = origin_disinherited_heir }

			limit = { has_origin = origin_restored_successor_government }
			log = "\\[this.GetName]"
		}
		log = "-------------------------"
	}

	every_playable_country = {
		limit = { has_civstrife_origin = yes }
		root = { change_variable = { which = total_civwar_countries value = 1 } }
	}
	log = "Total Civwar Countries: \\[this.total_civwar_countries]"
}

get_civwar_num = {
	every_playable_country = {
		limit = {
			OR = {
				is_variable_set = civwar_number
				any_owned_planet = {
					OR = {
						is_variable_set = reb_number
						is_variable_set = sec_reb_number
						is_variable_set = slave_reb_number
						is_variable_set = sec_slave_reb_number
					}
				}
			}
		}
		log = "\\[this.GetName]:"
		if = {
			limit = { is_variable_set = civwar_number }
			log = "\\[this.civwar_number] - Civil Wars"
		}
		if = {
			limit = {
				any_owned_planet = {
					OR = {
						is_variable_set = reb_number
						is_variable_set = sec_reb_number
						is_variable_set = slave_reb_number
						is_variable_set = sec_slave_reb_number
					}
				}
			}
			every_owned_planet = {
				limit = {
					OR = {
						is_variable_set = reb_number
						is_variable_set = sec_reb_number
						is_variable_set = slave_reb_number
						is_variable_set = sec_slave_reb_number
					}
				}
				log = "-------------"
				if = {
					limit = { is_variable_set = reb_number }
					log = "\\[this.GetName]: \\[this.reb_number] - Planetary Rebellions"
				}
				if = {
					limit = { is_variable_set = sec_reb_number }
					log = "\\[this.GetName]: \\[this.sec_reb_number] - Sectoral Rebellions"
				}
				if = {
					limit = { is_variable_set = slave_reb_number }
					log = "\\[this.GetName]: \\[this.slave_reb_number] - Planetary Slave Rebellions"
				}
				if = {
					limit = { is_variable_set = sec_slave_reb_number }
					log = "\\[this.GetName]: \\[this.sec_slave_reb_number] - Sectoral Slave Rebellions"
				}
				log = "-------------------------"
			}
		}
		log = "==============================="
	}
}

get_traitors_effect = {
	every_country = {
		limit = {
			OR = {
				any_owned_leader = { has_trait = leader_trait_traitor }
				any_owned_leader = { has_trait = leader_trait_traitor_heir }
			}
		}
		log = "Traitor HOST: \\[this.GetName]"
		if = {
			limit = { any_owned_leader = { has_trait = leader_trait_traitor } }
			every_owned_leader = {
				limit = { has_trait = leader_trait_traitor }
				log = "Normal Traitor: \\[this.GetName]"
			}
		}
		if = {
			limit = { any_owned_leader = { has_trait = leader_trait_traitor_heir } }
			every_owned_leader = {
				limit = { has_trait = leader_trait_traitor_heir }
				log = "Heir Traitor: \\[this.GetName]"
			}
		}
		log = "***************"
	}
	log = "===================="
}

}

test_distance_effect = {

	every_playable_country = {
		limit = {
			#is_country_type_with_subjects = yes
			is_country_type = default
			has_communications = root
			is_gestalt = no
			OR = {
				is_homicidal = no
				AND = {
					is_homicidal = yes
					event_target:defeated_traitor = { is_same_species = prev }
				}
				AND = {
					is_organic_zealots = yes
					event_target:defeated_traitor = { is_mechanical_species = no }
				}
			}
			NOR = {
				is_loyal_to = root
				is_overlord_to = root
				is_custodial_to = root
				is_same_value = root
				has_overlord = root
				has_country_flag = traitor_already_asked_@event_target:defeated_traitor
			}
		}

		set_variable = { which = distance_from_traitor value = 0 }

		save_event_target_as = potential_country

		root.capital_scope = {
			export_trigger_value_to_variable = {
				trigger = distance_to_empire
				parameters = { who = event_target:potential_country }
				variable = distance_from_traitor
			}
		}

		set_variable = { which = distance_from_traitor value = root.capital_scope.distance_from_traitor }
	}
}
